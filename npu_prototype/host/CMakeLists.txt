cmake_minimum_required(VERSION 3.17)
project(bm25_contribution_runner LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(TOOLKIT_HOME "/usr/local/Ascend/ascend-toolkit/latest" CACHE PATH "Actual CANN toolkit root")
find_path(ACL_INCLUDE acl/acl.h PATHS "${TOOLKIT_HOME}/include" NO_DEFAULT_PATH)
find_library(ACL_LIBRARY ascendcl PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
if(NOT ACL_INCLUDE OR NOT ACL_LIBRARY)
    message(FATAL_ERROR "CANN headers/library not found; set TOOLKIT_HOME to the actual toolkit root")
endif()
add_executable(bm25_contribution_runner contribution_runner.cpp)
target_include_directories(bm25_contribution_runner PRIVATE "${ACL_INCLUDE}")
target_link_libraries(bm25_contribution_runner PRIVATE "${ACL_LIBRARY}")
target_compile_options(bm25_contribution_runner PRIVATE -Wall -Wextra -O2)
set_target_properties(bm25_contribution_runner PROPERTIES BUILD_RPATH "${TOOLKIT_HOME}/lib64")
