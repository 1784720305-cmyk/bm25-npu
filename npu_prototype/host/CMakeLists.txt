cmake_minimum_required(VERSION 3.17)
project(bm25_aclnn_runner LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(TOOLKIT_HOME "/usr/local/Ascend/cann-9.2.0" CACHE PATH "Actual CANN toolkit root")
set(BM25_OPP_ROOT "" CACHE PATH "Installed vendor root containing the selected operator's op_api")
set(BM25_RUNNER "contribution" CACHE STRING "Operator to build: contribution or accumulate")
set_property(CACHE BM25_RUNNER PROPERTY STRINGS contribution accumulate)
if(NOT BM25_RUNNER STREQUAL "contribution" AND NOT BM25_RUNNER STREQUAL "accumulate")
    message(FATAL_ERROR "BM25_RUNNER must be contribution or accumulate")
endif()
if(NOT BM25_OPP_ROOT)
    message(FATAL_ERROR "Set BM25_OPP_ROOT to the installed vendor directory for ${BM25_RUNNER}")
endif()
find_path(ACL_INCLUDE acl/acl.h PATHS "${TOOLKIT_HOME}/include" NO_DEFAULT_PATH)
find_path(ACL_META_INCLUDE aclnn/acl_meta.h PATHS "${TOOLKIT_HOME}/include" "${TOOLKIT_HOME}/include/aclnnop" NO_DEFAULT_PATH)
find_path(BM25_API_INCLUDE aclnn_bm25_${BM25_RUNNER}.h PATHS "${BM25_OPP_ROOT}/op_api/include" NO_DEFAULT_PATH)
find_library(ACL_LIBRARY ascendcl PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
find_library(NNOP_LIBRARY nnopbase PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
find_library(BM25_API_LIBRARY cust_opapi PATHS "${BM25_OPP_ROOT}/op_api/lib" NO_DEFAULT_PATH)
if(NOT ACL_INCLUDE OR NOT ACL_META_INCLUDE OR NOT ACL_LIBRARY OR NOT NNOP_LIBRARY OR NOT BM25_API_INCLUDE OR NOT BM25_API_LIBRARY)
    message(FATAL_ERROR "CANN or installed generated BM25 header/library missing; check TOOLKIT_HOME and BM25_OPP_ROOT")
endif()
# Use separate build directories for each operator: both vendor API libraries share a filename.
set(RUNNER_TARGET bm25_${BM25_RUNNER}_runner)
add_executable(${RUNNER_TARGET} ${BM25_RUNNER}_runner.cpp)
target_include_directories(${RUNNER_TARGET} PRIVATE "${ACL_INCLUDE}" "${ACL_META_INCLUDE}" "${BM25_API_INCLUDE}")
target_link_libraries(${RUNNER_TARGET} PRIVATE "${BM25_API_LIBRARY}" "${NNOP_LIBRARY}" "${ACL_LIBRARY}")
target_compile_options(${RUNNER_TARGET} PRIVATE -Wall -Wextra -O2)
set_target_properties(${RUNNER_TARGET} PROPERTIES BUILD_RPATH "${BM25_OPP_ROOT}/op_api/lib;${TOOLKIT_HOME}/lib64")
message(STATUS "BM25 runner: ${BM25_RUNNER}")
message(STATUS "BM25 generated header: ${BM25_API_INCLUDE}/aclnn_bm25_${BM25_RUNNER}.h")
message(STATUS "BM25 generated API library: ${BM25_API_LIBRARY}")
