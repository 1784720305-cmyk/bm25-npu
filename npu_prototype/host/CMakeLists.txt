cmake_minimum_required(VERSION 3.17)
project(bm25_contribution_runner LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(TOOLKIT_HOME "/usr/local/Ascend/cann-9.2.0" CACHE PATH "Actual CANN toolkit root")
set(BM25_OPP_ROOT "" CACHE PATH "Installed vendor root containing op_api, e.g. results/custom_opp/vendors/bm25")
if(NOT BM25_OPP_ROOT)
    message(FATAL_ERROR "Set BM25_OPP_ROOT to the installed bm25 vendor directory")
endif()
find_path(ACL_INCLUDE acl/acl.h PATHS "${TOOLKIT_HOME}/include" NO_DEFAULT_PATH)
find_path(ACL_META_INCLUDE aclnn/acl_meta.h PATHS "${TOOLKIT_HOME}/include" "${TOOLKIT_HOME}/include/aclnnop" NO_DEFAULT_PATH)
find_path(BM25_API_INCLUDE aclnn_bm25_contribution.h PATHS "${BM25_OPP_ROOT}/op_api/include" NO_DEFAULT_PATH)
find_library(ACL_LIBRARY ascendcl PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
find_library(NNOP_LIBRARY nnopbase PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
find_library(BM25_API_LIBRARY cust_opapi PATHS "${BM25_OPP_ROOT}/op_api/lib" NO_DEFAULT_PATH)
if(NOT ACL_INCLUDE OR NOT ACL_META_INCLUDE OR NOT ACL_LIBRARY OR NOT NNOP_LIBRARY OR NOT BM25_API_INCLUDE OR NOT BM25_API_LIBRARY)
    message(FATAL_ERROR "CANN or installed generated BM25 header/library missing; check TOOLKIT_HOME and BM25_OPP_ROOT")
endif()
add_executable(bm25_contribution_runner contribution_runner.cpp)
target_include_directories(bm25_contribution_runner PRIVATE "${ACL_INCLUDE}" "${ACL_META_INCLUDE}" "${BM25_API_INCLUDE}")
target_link_libraries(bm25_contribution_runner PRIVATE "${BM25_API_LIBRARY}" "${NNOP_LIBRARY}" "${ACL_LIBRARY}")
target_compile_options(bm25_contribution_runner PRIVATE -Wall -Wextra -O2)
set_target_properties(bm25_contribution_runner PROPERTIES BUILD_RPATH "${BM25_OPP_ROOT}/op_api/lib;${TOOLKIT_HOME}/lib64")
message(STATUS "BM25 generated header: ${BM25_API_INCLUDE}/aclnn_bm25_contribution.h")
message(STATUS "BM25 generated API library: ${BM25_API_LIBRARY}")
