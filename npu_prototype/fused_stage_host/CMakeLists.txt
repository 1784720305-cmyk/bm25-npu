cmake_minimum_required(VERSION 3.17)
project(bm25_fused_stage_diagnostic LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
 message(FATAL_ERROR "Requires actual A5 Linux/CANN headers")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(TOOLKIT_HOME "/usr/local/Ascend/cann-9.2.0" CACHE PATH "Actual CANN installation")
set(BM25_VENDOR_DIR "" CACHE PATH "Diagnostic runtime aliases")
find_path(ACL_INCLUDE acl/acl.h PATHS "${TOOLKIT_HOME}/include" NO_DEFAULT_PATH)
find_path(META_INCLUDE aclnn/acl_meta.h PATHS "${TOOLKIT_HOME}/include" "${TOOLKIT_HOME}/include/aclnnop" NO_DEFAULT_PATH)
find_library(ACL_LIBRARY ascendcl PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
find_library(NNOP_LIBRARY nnopbase PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
if(NOT ACL_INCLUDE OR NOT META_INCLUDE OR NOT ACL_LIBRARY OR NOT NNOP_LIBRARY OR NOT BM25_VENDOR_DIR)
 message(FATAL_ERROR "Actual CANN headers/libraries and diagnostic vendor required")
endif()
set(API_INCLUDES "")
foreach(pair bm25_fused_score_topk/aclnn_bm25_fused_score_top_k.h bm25_topk_merge/aclnn_bm25_top_k_merge.h)
 string(REPLACE "/" "/op_api/include/" header "${pair}")
 if(NOT EXISTS "${BM25_VENDOR_DIR}/${header}")
  message(FATAL_ERROR "Missing actual generated header: ${header}")
 endif()
 get_filename_component(parent "${BM25_VENDOR_DIR}/${header}" DIRECTORY)
 list(APPEND API_INCLUDES "${parent}")
endforeach()
add_executable(bm25_fused_stage_diagnostic stage_runner.cpp)
target_include_directories(bm25_fused_stage_diagnostic PRIVATE "${ACL_INCLUDE}" "${META_INCLUDE}" ${API_INCLUDES}
 "${CMAKE_CURRENT_SOURCE_DIR}/../fused_chain_host")
target_link_libraries(bm25_fused_stage_diagnostic PRIVATE "${NNOP_LIBRARY}" "${ACL_LIBRARY}" ${CMAKE_DL_LIBS})
target_compile_options(bm25_fused_stage_diagnostic PRIVATE -Wall -Wextra -O2 -fno-fast-math -ffp-contract=off)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/stage_runner.cpp" HOST_SHA)
target_compile_definitions(bm25_fused_stage_diagnostic PRIVATE BM25_STAGE_HOST_SHA="${HOST_SHA}")
set_target_properties(bm25_fused_stage_diagnostic PROPERTIES BUILD_RPATH "${TOOLKIT_HOME}/lib64")
