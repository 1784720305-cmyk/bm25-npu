cmake_minimum_required(VERSION 3.17)
project(bm25_pipeline_runner LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This ACLNN pipeline loader targets the actual A5 Linux host")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(TOOLKIT_HOME "/usr/local/Ascend/cann-9.2.0" CACHE PATH "Actual CANN toolkit root")
set(BM25_VENDOR_DIR "" CACHE PATH "Installed parent vendors directory containing bm25, bm25_accumulate, bm25_topk")
if(NOT BM25_VENDOR_DIR)
    message(FATAL_ERROR "Set BM25_VENDOR_DIR to results/custom_opp/vendors")
endif()
find_path(ACL_INCLUDE acl/acl.h PATHS "${TOOLKIT_HOME}/include" NO_DEFAULT_PATH)
find_path(ACL_META_INCLUDE aclnn/acl_meta.h PATHS "${TOOLKIT_HOME}/include" "${TOOLKIT_HOME}/include/aclnnop" NO_DEFAULT_PATH)
find_library(ACL_LIBRARY ascendcl PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
find_library(NNOP_LIBRARY nnopbase PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
if(NOT ACL_INCLUDE OR NOT ACL_META_INCLUDE OR NOT ACL_LIBRARY OR NOT NNOP_LIBRARY)
    message(FATAL_ERROR "CANN headers/runtime missing; check TOOLKIT_HOME")
endif()
find_path(CONTRIBUTION_INCLUDE aclnn_bm25_contribution.h PATHS "${BM25_VENDOR_DIR}/bm25/op_api/include" NO_DEFAULT_PATH)
find_path(ACCUMULATE_INCLUDE aclnn_bm25_accumulate.h PATHS "${BM25_VENDOR_DIR}/bm25_accumulate/op_api/include" NO_DEFAULT_PATH)
find_path(TOPK_INCLUDE aclnn_bm25_top_k.h PATHS "${BM25_VENDOR_DIR}/bm25_topk/op_api/include" NO_DEFAULT_PATH)
if(NOT CONTRIBUTION_INCLUDE OR NOT ACCUMULATE_INCLUDE OR NOT TOPK_INCLUDE)
    message(FATAL_ERROR "Actual generated headers from all three installed vendors are required")
endif()
foreach(vendor bm25 bm25_accumulate bm25_topk)
    if(NOT EXISTS "${BM25_VENDOR_DIR}/${vendor}/op_api/lib/libcust_opapi.so")
        message(FATAL_ERROR "Installed API library missing for ${vendor}")
    endif()
endforeach()
add_executable(bm25_pipeline_runner pipeline_runner.cpp)
target_include_directories(bm25_pipeline_runner PRIVATE "${ACL_INCLUDE}" "${ACL_META_INCLUDE}"
    "${CONTRIBUTION_INCLUDE}" "${ACCUMULATE_INCLUDE}" "${TOPK_INCLUDE}")
# Do not link the three identically-named cust_opapi libraries: load their absolute paths separately.
target_link_libraries(bm25_pipeline_runner PRIVATE "${NNOP_LIBRARY}" "${ACL_LIBRARY}" ${CMAKE_DL_LIBS})
target_compile_options(bm25_pipeline_runner PRIVATE -Wall -Wextra -O2)
set_target_properties(bm25_pipeline_runner PROPERTIES BUILD_RPATH "${TOOLKIT_HOME}/lib64")
message(STATUS "Pipeline generated headers: ${CONTRIBUTION_INCLUDE};${ACCUMULATE_INCLUDE};${TOPK_INCLUDE}")
message(STATUS "Runtime API library parent: ${BM25_VENDOR_DIR}; absolute-path loading and symbol-owner checks required")
