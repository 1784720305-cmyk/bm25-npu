cmake_minimum_required(VERSION 3.17)
project(bm25_fused_chain_diagnostic LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Build on A5 Linux with actual installed generated fused/merge headers")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(TOOLKIT_HOME "/usr/local/Ascend/cann-9.2.0" CACHE PATH "Actual CANN root")
set(BM25_VENDOR_DIR "" CACHE PATH "Actual installed vendors parent")
find_path(ACL_INCLUDE acl/acl.h PATHS "${TOOLKIT_HOME}/include" NO_DEFAULT_PATH)
find_path(META_INCLUDE aclnn/acl_meta.h PATHS "${TOOLKIT_HOME}/include" "${TOOLKIT_HOME}/include/aclnnop" NO_DEFAULT_PATH)
find_library(ACL_LIBRARY ascendcl PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
find_library(NNOP_LIBRARY nnopbase PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
if(NOT ACL_INCLUDE OR NOT META_INCLUDE OR NOT ACL_LIBRARY OR NOT NNOP_LIBRARY OR NOT BM25_VENDOR_DIR)
    message(FATAL_ERROR "Actual CANN headers/libraries or BM25_VENDOR_DIR missing")
endif()
set(API_INCLUDES "")
foreach(pair bm25_fused_score_topk/aclnn_bm25_fused_score_top_k.h bm25_topk_merge/aclnn_bm25_top_k_merge.h)
    string(REPLACE "/" "/op_api/include/" header "${pair}")
    if(NOT EXISTS "${BM25_VENDOR_DIR}/${header}")
        message(FATAL_ERROR "Actual generated header missing: ${header}")
    endif()
    get_filename_component(parent "${BM25_VENDOR_DIR}/${header}" DIRECTORY)
    list(APPEND API_INCLUDES "${parent}")
    message(STATUS "Actual generated header: ${BM25_VENDOR_DIR}/${header}")
endforeach()
foreach(vendor bm25_fused_score_topk bm25_topk_merge)
    if(NOT EXISTS "${BM25_VENDOR_DIR}/${vendor}/op_api/lib/libcust_opapi.so")
        message(FATAL_ERROR "Installed API library missing: ${vendor}")
    endif()
endforeach()
add_executable(bm25_fused_chain_runner fused_chain_runner.cpp)
target_include_directories(bm25_fused_chain_runner PRIVATE "${ACL_INCLUDE}" "${META_INCLUDE}" ${API_INCLUDES})
target_link_libraries(bm25_fused_chain_runner PRIVATE "${NNOP_LIBRARY}" "${ACL_LIBRARY}" ${CMAKE_DL_LIBS})
target_compile_options(bm25_fused_chain_runner PRIVATE -Wall -Wextra -O2 -fno-fast-math -ffp-contract=off)
foreach(source CMakeLists.txt fused_runtime.h fused_chain_runner.cpp)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
endforeach()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" CMAKE_SHA)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/fused_runtime.h" RUNTIME_SHA)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/fused_chain_runner.cpp" RUNNER_SHA)
target_compile_definitions(bm25_fused_chain_runner PRIVATE BM25_FUSED_CMAKE_SHA="${CMAKE_SHA}"
    BM25_FUSED_RUNTIME_SHA="${RUNTIME_SHA}" BM25_FUSED_RUNNER_SHA="${RUNNER_SHA}")
set_target_properties(bm25_fused_chain_runner PROPERTIES BUILD_RPATH "${TOOLKIT_HOME}/lib64")
message(STATUS "Reviewed fused ABI: seven inputs, k/shards/debug, three outputs; two absolute API libraries")
message(STATUS "A/B/A/empty/B x debug1/debug0; S64 uses two existing merge stages; diagnostic only, no timing or CPU fallback")
