cmake_minimum_required(VERSION 3.17)
project(bm25_topk_positive_compact_diagnostic LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Build on the A5 Linux host with the actual installed generated header")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(TOOLKIT_HOME "/usr/local/Ascend/cann-9.2.0" CACHE PATH "Actual CANN toolkit root")
set(BM25_OPP_ROOT "" CACHE PATH "Installed bm25_topk_positive_compact vendor root")
get_filename_component(VENDOR_NAME "${BM25_OPP_ROOT}" NAME)
if(NOT VENDOR_NAME STREQUAL "bm25_topk_positive_compact")
    message(FATAL_ERROR "BM25_OPP_ROOT must select the independent bm25_topk_positive_compact vendor")
endif()
find_path(ACL_INCLUDE acl/acl.h PATHS "${TOOLKIT_HOME}/include" NO_DEFAULT_PATH)
find_path(META_INCLUDE aclnn/acl_meta.h PATHS "${TOOLKIT_HOME}/include" "${TOOLKIT_HOME}/include/aclnnop" NO_DEFAULT_PATH)
find_path(API_INCLUDE aclnn_bm25_top_k.h PATHS "${BM25_OPP_ROOT}/op_api/include" NO_DEFAULT_PATH)
find_library(ACL_LIBRARY ascendcl PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
find_library(NNOP_LIBRARY nnopbase PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
if(NOT ACL_INCLUDE OR NOT META_INCLUDE OR NOT API_INCLUDE OR NOT ACL_LIBRARY OR NOT NNOP_LIBRARY)
    message(FATAL_ERROR "Actual CANN or installed Radix generated header missing; use a fresh build directory")
endif()
if(NOT EXISTS "${BM25_OPP_ROOT}/op_api/lib/libcust_opapi.so")
    message(FATAL_ERROR "Installed Radix API library missing")
endif()
add_executable(bm25_topk_positive_compact_runner positive_compact_runner.cpp)
target_include_directories(bm25_topk_positive_compact_runner PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../topk_radix_host" "${ACL_INCLUDE}" "${META_INCLUDE}" "${API_INCLUDE}")
target_link_libraries(bm25_topk_positive_compact_runner PRIVATE "${NNOP_LIBRARY}" "${ACL_LIBRARY}" ${CMAKE_DL_LIBS})
target_compile_options(bm25_topk_positive_compact_runner PRIVATE -Wall -Wextra -O2 -fno-fast-math -ffp-contract=off)
set_target_properties(bm25_topk_positive_compact_runner PROPERTIES BUILD_RPATH "${TOOLKIT_HOME}/lib64")
message(STATUS "Actual positive-compaction header: ${API_INCLUDE}/aclnn_bm25_top_k.h")
message(STATUS "Standalone Top-K diagnostic; no contribution/accumulation headers or other custom API libraries")
message(STATUS "Loads new API by absolute path; checks both symbol owners before any device operation")

# Embed the source bytes compiled into this diagnostic; old accepted runners remain unchanged.
get_filename_component(PACKAGE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(HOST_SOURCE_PATHS npu_prototype/topk_positive_compact_host/CMakeLists.txt
    npu_prototype/topk_positive_compact_host/positive_compact_runner.cpp
    npu_prototype/topk_radix_host/radix_host_common.h)
set(HOST_JSON "{")
foreach(SOURCE IN LISTS HOST_SOURCE_PATHS)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PACKAGE_ROOT}/${SOURCE}")
    file(SHA256 "${PACKAGE_ROOT}/${SOURCE}" SOURCE_SHA)
    string(APPEND HOST_JSON "\"${SOURCE}\":\"${SOURCE_SHA}\",")
endforeach()
string(REGEX REPLACE ",$" "}" HOST_JSON "${HOST_JSON}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/positive_compact_build_info.h"
    "#pragma once\n#define BM25_HOST_SOURCES_JSON R\"bm25(${HOST_JSON})bm25\"\n")
