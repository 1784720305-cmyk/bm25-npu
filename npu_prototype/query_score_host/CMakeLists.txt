cmake_minimum_required(VERSION 3.17)
project(bm25_query_score_session LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Build on A5 Linux using the actually installed generated header")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(TOOLKIT_HOME "/usr/local/Ascend/cann-9.2.0" CACHE PATH "Actual CANN toolkit root")
set(BM25_OPP_ROOT "" CACHE PATH "Installed independent bm25_query_score vendor")
get_filename_component(VENDOR_NAME "${BM25_OPP_ROOT}" NAME)
if(NOT VENDOR_NAME STREQUAL "bm25_query_score")
    message(FATAL_ERROR "BM25_OPP_ROOT must select bm25_query_score")
endif()
find_path(ACL_INCLUDE acl/acl.h PATHS "${TOOLKIT_HOME}/include" NO_DEFAULT_PATH)
find_path(META_INCLUDE aclnn/acl_meta.h PATHS "${TOOLKIT_HOME}/include" "${TOOLKIT_HOME}/include/aclnnop" NO_DEFAULT_PATH)
find_path(API_INCLUDE aclnn_bm25_query_score.h PATHS "${BM25_OPP_ROOT}/op_api/include" NO_DEFAULT_PATH)
find_library(ACL_LIBRARY ascendcl PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
find_library(NNOP_LIBRARY nnopbase PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
if(NOT ACL_INCLUDE OR NOT META_INCLUDE OR NOT API_INCLUDE OR NOT ACL_LIBRARY OR NOT NNOP_LIBRARY)
    message(FATAL_ERROR "Actual CANN or generated query-score header missing; check the paths")
endif()
if(NOT EXISTS "${BM25_OPP_ROOT}/op_api/lib/libcust_opapi.so")
    message(FATAL_ERROR "Installed query-score API library missing")
endif()
add_executable(bm25_query_score_runner query_score_runner.cpp)
target_include_directories(bm25_query_score_runner PRIVATE "${ACL_INCLUDE}" "${META_INCLUDE}" "${API_INCLUDE}")
target_link_libraries(bm25_query_score_runner PRIVATE "${NNOP_LIBRARY}" "${ACL_LIBRARY}" ${CMAKE_DL_LIBS})
target_compile_options(bm25_query_score_runner PRIVATE -Wall -Wextra -O2 -fno-fast-math -ffp-contract=off)
foreach(SOURCE_FILE IN ITEMS CMakeLists.txt query_score_runtime.h query_score_runner.cpp)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${SOURCE_FILE}")
endforeach()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" CMAKE_SHA)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/query_score_runtime.h" RUNTIME_SHA)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/query_score_runner.cpp" RUNNER_SHA)
target_compile_definitions(bm25_query_score_runner PRIVATE
    BM25_CMAKE_SHA256="${CMAKE_SHA}" BM25_RUNTIME_SHA256="${RUNTIME_SHA}" BM25_RUNNER_SHA256="${RUNNER_SHA}")
set_target_properties(bm25_query_score_runner PROPERTIES BUILD_RPATH "${TOOLKIT_HOME}/lib64")
message(STATUS "Actual generated header: ${API_INCLUDE}/aclnn_bm25_query_score.h")
message(STATUS "Only query-score API, loaded by absolute path with symbol-owner checks")
message(STATUS "Resident-index score correctness diagnostic, not a Top-K or search benchmark")
