cmake_minimum_required(VERSION 3.17)
project(bm25_fused_public_search LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Build on A5 Linux with actual installed generated fused/merge headers")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(TOOLKIT_HOME "/usr/local/Ascend/cann-9.2.0" CACHE PATH "Actual CANN root")
set(BM25_VENDOR_DIR "" CACHE PATH "Actual installed vendors parent")
find_path(ACL_INCLUDE acl/acl.h PATHS "${TOOLKIT_HOME}/include" NO_DEFAULT_PATH)
find_path(META_INCLUDE aclnn/acl_meta.h PATHS "${TOOLKIT_HOME}/include" "${TOOLKIT_HOME}/include/aclnnop" NO_DEFAULT_PATH)
find_library(ACL_LIBRARY ascendcl PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
find_library(NNOP_LIBRARY nnopbase PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
if(NOT ACL_INCLUDE OR NOT META_INCLUDE OR NOT ACL_LIBRARY OR NOT NNOP_LIBRARY OR NOT BM25_VENDOR_DIR)
    message(FATAL_ERROR "Actual CANN headers/libraries or BM25_VENDOR_DIR missing")
endif()
set(API_INCLUDES "")
foreach(pair bm25_fused_score_topk/aclnn_bm25_fused_score_top_k.h bm25_topk_merge/aclnn_bm25_top_k_merge.h)
    string(REPLACE "/" "/op_api/include/" header "${pair}")
    if(NOT EXISTS "${BM25_VENDOR_DIR}/${header}")
        message(FATAL_ERROR "Actual generated header missing: ${header}")
    endif()
    get_filename_component(parent "${BM25_VENDOR_DIR}/${header}" DIRECTORY)
    list(APPEND API_INCLUDES "${parent}")
    message(STATUS "Actual generated header: ${BM25_VENDOR_DIR}/${header}")
endforeach()
foreach(vendor bm25_fused_score_topk bm25_topk_merge)
    if(NOT EXISTS "${BM25_VENDOR_DIR}/${vendor}/op_api/lib/libcust_opapi.so")
        message(FATAL_ERROR "Installed API library missing: ${vendor}")
    endif()
endforeach()
add_executable(bm25_fused_search_benchmark fused_benchmark.cpp)
target_include_directories(bm25_fused_search_benchmark PRIVATE "${ACL_INCLUDE}" "${META_INCLUDE}" ${API_INCLUDES} "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(bm25_fused_search_benchmark PRIVATE "${NNOP_LIBRARY}" "${ACL_LIBRARY}" ${CMAKE_DL_LIBS})
target_compile_options(bm25_fused_search_benchmark PRIVATE -Wall -Wextra -O2 -fno-fast-math -ffp-contract=off)
set(SOURCES npu_prototype/fused_search_host/CMakeLists.txt npu_prototype/fused_search_host/fused_search.h
 npu_prototype/fused_search_host/fused_api.h npu_prototype/fused_search_host/fused_benchmark.cpp
 npu_prototype/fused_chain_host/fused_runtime.h npu_prototype/search_host/search_inputs.h
 npu_prototype/search_benchmark_host/benchmark_common.h)
set(RECORD "#pragma once\n#include <cstddef>\nstatic constexpr const char *BM25_BENCH_SOURCES[][2] = {\n")
foreach(source IN LISTS SOURCES)
 set(path "${CMAKE_CURRENT_SOURCE_DIR}/../../${source}")
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
 file(SHA256 "${path}" sha)
 string(APPEND RECORD "{\"${source}\",\"${sha}\"},\n")
endforeach()
string(APPEND RECORD "};\nstatic constexpr std::size_t BM25_BENCH_SOURCE_COUNT=sizeof(BM25_BENCH_SOURCES)/sizeof(BM25_BENCH_SOURCES[0]);\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/compiled_benchmark_sources.h" "${RECORD}")
set_target_properties(bm25_fused_search_benchmark PROPERTIES BUILD_RPATH "${TOOLKIT_HOME}/lib64")
include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_INCLUDES "${ACL_INCLUDE}")
set(CMAKE_REQUIRED_LIBRARIES "${ACL_LIBRARY}")
check_cxx_source_compiles("#include <acl/acl.h>
int main(){int64_t v=0;aclrtGetDeviceInfo(0,ACL_DEV_ATTR_VECTOR_CORE_NUM,&v);
return int(ACL_DEV_ATTR_WARP_SIZE)+int(ACL_DEV_ATTR_MAX_THREAD_PER_VECTOR_CORE)+int(ACL_DEV_ATTR_UBUF_PER_VECTOR_CORE);}" BM25_HAS_DEVICE_ATTRIBUTES)
if(BM25_HAS_DEVICE_ATTRIBUTES)
 add_executable(bm25_hardware_query hardware_query.cpp)
 target_include_directories(bm25_hardware_query PRIVATE "${ACL_INCLUDE}")
 target_link_libraries(bm25_hardware_query PRIVATE "${ACL_LIBRARY}")
 set_target_properties(bm25_hardware_query PROPERTIES BUILD_RPATH "${TOOLKIT_HOME}/lib64")
else()
 message(STATUS "Actual installed headers/library do not expose reviewed hardware query; mark unavailable, no environment changes")
endif()
