cmake_minimum_required(VERSION 3.17)
project(bm25_official_topk_reference LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Build on the A5 Linux host with the actual CANN headers")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(TOOLKIT_HOME "/usr/local/Ascend/cann-9.2.0" CACHE PATH "Actual CANN installation")
find_path(ACL_INCLUDE acl/acl.h PATHS "${TOOLKIT_HOME}/include" NO_DEFAULT_PATH)
find_path(OP_INCLUDE aclnn_topk.h PATHS "${TOOLKIT_HOME}/include/aclnnop" NO_DEFAULT_PATH)
find_path(META_INCLUDE aclnn/acl_meta.h PATHS "${TOOLKIT_HOME}/include" "${TOOLKIT_HOME}/include/aclnnop" NO_DEFAULT_PATH)
find_library(ACL_LIBRARY ascendcl PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
find_library(NNOP_LIBRARY nnopbase PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
if(NOT ACL_INCLUDE OR NOT OP_INCLUDE OR NOT META_INCLUDE OR NOT ACL_LIBRARY OR NOT NNOP_LIBRARY
   OR NOT EXISTS "${TOOLKIT_HOME}/lib64/libopapi.so")
    message(FATAL_ERROR "Actual CANN headers or libraries missing; check TOOLKIT_HOME")
endif()
add_executable(bm25_official_topk_runner official_topk_runner.cpp)
target_include_directories(bm25_official_topk_runner PRIVATE "${ACL_INCLUDE}" "${OP_INCLUDE}" "${META_INCLUDE}")
target_link_libraries(bm25_official_topk_runner PRIVATE "${NNOP_LIBRARY}" "${ACL_LIBRARY}" ${CMAKE_DL_LIBS})
target_compile_options(bm25_official_topk_runner PRIVATE -Wall -Wextra -O2 -ffp-contract=off)
set_target_properties(bm25_official_topk_runner PROPERTIES BUILD_RPATH "${TOOLKIT_HOME}/lib64")
message(STATUS "Official header: ${OP_INCLUDE}/aclnn_topk.h")
message(STATUS "Official API loaded by absolute path at runtime: ${TOOLKIT_HOME}/lib64/libopapi.so")
message(STATUS "Reference only: no custom operator, ATC, OM, Faiss or IndexSDK")
