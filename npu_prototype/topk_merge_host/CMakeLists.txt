cmake_minimum_required(VERSION 3.17)
project(bm25_topk_merge_diagnostic LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Build on the A5 Linux host with the installed generated header")
endif()
set(TOOLKIT_HOME "/usr/local/Ascend/cann-9.2.0" CACHE PATH "Actual toolkit root")
set(BM25_MERGE_OPP_ROOT "" CACHE PATH "Installed bm25_topk_merge vendor")
find_path(ACL_INCLUDE acl/acl.h PATHS "${TOOLKIT_HOME}/include" NO_DEFAULT_PATH)
find_path(META_INCLUDE aclnn/acl_meta.h PATHS "${TOOLKIT_HOME}/include" "${TOOLKIT_HOME}/include/aclnnop" NO_DEFAULT_PATH)
find_library(ACL_LIBRARY ascendcl PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
find_library(NNOP_LIBRARY nnopbase PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
find_library(DL_LIBRARY dl REQUIRED)
if(NOT ACL_INCLUDE OR NOT META_INCLUDE OR NOT ACL_LIBRARY OR NOT NNOP_LIBRARY OR
   NOT EXISTS "${BM25_MERGE_OPP_ROOT}/op_api/include/aclnn_bm25_top_k_merge.h" OR
   NOT EXISTS "${BM25_MERGE_OPP_ROOT}/op_api/lib/libcust_opapi.so")
    message(FATAL_ERROR "Missing actual CANN or installed merge header/library; check both root paths")
endif()
add_executable(bm25_topk_merge_runner merge_runner.cpp)
target_include_directories(bm25_topk_merge_runner PRIVATE "${ACL_INCLUDE}" "${META_INCLUDE}" "${BM25_MERGE_OPP_ROOT}/op_api/include")
target_link_libraries(bm25_topk_merge_runner PRIVATE "${NNOP_LIBRARY}" "${ACL_LIBRARY}" "${DL_LIBRARY}")
target_compile_options(bm25_topk_merge_runner PRIVATE -Wall -Wextra -O2)
set_target_properties(bm25_topk_merge_runner PROPERTIES BUILD_RPATH "${TOOLKIT_HOME}/lib64")
message(STATUS "Actual merge header: ${BM25_MERGE_OPP_ROOT}/op_api/include/aclnn_bm25_top_k_merge.h")
message(STATUS "Merge API is loaded by absolute path and both symbol owners are checked at runtime")
