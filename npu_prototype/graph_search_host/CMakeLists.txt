cmake_minimum_required(VERSION 3.17)
project(bm25_planned_search LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(TOOLKIT_HOME "/usr/local/Ascend/cann-9.2.0" CACHE PATH "Actual CANN installation")
set(BM25_VENDOR_DIR "" CACHE PATH "Runtime aliases to accepted vendor or directory vendor")
find_package(OpenMP REQUIRED COMPONENTS CXX)
find_path(ACL_INCLUDE acl/acl.h PATHS "${TOOLKIT_HOME}/include" NO_DEFAULT_PATH)
find_path(META_INCLUDE aclnn/acl_meta.h PATHS "${TOOLKIT_HOME}/include" "${TOOLKIT_HOME}/include/aclnnop" NO_DEFAULT_PATH)
find_library(ACL_LIBRARY ascendcl PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
find_library(NNOP_LIBRARY nnopbase PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
if(NOT ACL_INCLUDE OR NOT META_INCLUDE OR NOT ACL_LIBRARY OR NOT NNOP_LIBRARY OR NOT BM25_VENDOR_DIR)
 message(FATAL_ERROR "Actual CANN headers/libraries and BM25_VENDOR_DIR required")
endif()
include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_INCLUDES "${ACL_INCLUDE}")
set(CMAKE_REQUIRED_LIBRARIES "${ACL_LIBRARY}")
check_cxx_source_compiles("#include <acl/acl.h>
int main(){aclrtStream s{};aclmdlRI m{};
aclmdlRICaptureBegin(s,ACL_MODEL_RI_CAPTURE_MODE_GLOBAL);aclmdlRICaptureEnd(s,&m);
aclmdlRIExecuteAsync(m,s);return aclmdlRIDestroy(m);}" BM25_ACTUAL_CAPTURE_API)
if(NOT BM25_ACTUAL_CAPTURE_API)
 message(FATAL_ERROR "Installed capture ABI unavailable; stop without fallback or benchmark")
endif()
set(API_INCLUDES "")
foreach(pair bm25_fused_score_topk/aclnn_bm25_fused_score_top_k.h bm25_topk_merge/aclnn_bm25_top_k_merge.h)
 string(REPLACE "/" "/op_api/include/" header "${pair}")
 if(NOT EXISTS "${BM25_VENDOR_DIR}/${header}")
  message(FATAL_ERROR "Actual generated API header missing: ${header}")
 endif()
 get_filename_component(parent "${BM25_VENDOR_DIR}/${header}" DIRECTORY)
 list(APPEND API_INCLUDES "${parent}")
endforeach()
set(SOURCES
 common/fixed_query_packet.h common/shard_directory.h
 cpu_reference/directory_search/cpu_directory_search.h cpu_reference/index_search/cpu_index_search.h
 cpu_reference/topk_document_shards.h npu_prototype/search_host/search_inputs.h
 npu_prototype/fused_chain_host/fused_runtime.h npu_prototype/fused_search_host/fused_api.h
 npu_prototype/search_benchmark_host/benchmark_common.h
 npu_prototype/graph_search_host/CMakeLists.txt npu_prototype/graph_search_host/planned_search.h
 npu_prototype/graph_search_host/planned_benchmark.cpp npu_prototype/graph_search_host/plan_check.cpp
 npu_prototype/graph_search_host/portable_check.cpp npu_prototype/graph_search_host/cpu_directory_benchmark.cpp)
set(RECORD "#pragma once\n#include <cstddef>\nstatic constexpr const char *BM25_BENCH_SOURCES[][2] = {\n")
foreach(source IN LISTS SOURCES)
 set(path "${CMAKE_CURRENT_SOURCE_DIR}/../../${source}")
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
 file(SHA256 "${path}" sha)
 string(APPEND RECORD "{\"${source}\",\"${sha}\"},\n")
endforeach()
string(APPEND RECORD "};\nstatic constexpr std::size_t BM25_BENCH_SOURCE_COUNT=sizeof(BM25_BENCH_SOURCES)/sizeof(BM25_BENCH_SOURCES[0]);\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/compiled_benchmark_sources.h" "${RECORD}")
foreach(target bm25_planned_search_benchmark bm25_plan_check)
 if(target STREQUAL "bm25_plan_check")
  set(source plan_check.cpp)
 else()
  set(source planned_benchmark.cpp)
 endif()
 add_executable(${target} ${source})
 target_include_directories(${target} PRIVATE "${ACL_INCLUDE}" "${META_INCLUDE}" ${API_INCLUDES} "${CMAKE_CURRENT_BINARY_DIR}")
 target_link_libraries(${target} PRIVATE "${NNOP_LIBRARY}" "${ACL_LIBRARY}" ${CMAKE_DL_LIBS})
 target_compile_options(${target} PRIVATE -Wall -Wextra -O2 -fno-fast-math -ffp-contract=off)
 set_target_properties(${target} PROPERTIES BUILD_RPATH "${TOOLKIT_HOME}/lib64")
endforeach()
add_executable(bm25_directory_cpu_benchmark cpu_directory_benchmark.cpp)
add_executable(bm25_plan_portable_check portable_check.cpp)
foreach(target bm25_directory_cpu_benchmark bm25_plan_portable_check)
 target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
 target_link_libraries(${target} PRIVATE OpenMP::OpenMP_CXX)
 target_compile_options(${target} PRIVATE -Wall -Wextra -O3 -march=native -fno-fast-math -ffp-contract=off)
endforeach()
