cmake_minimum_required(VERSION 3.17)
project(bm25_sharded_benchmark LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Compile this direct ACLNN experiment on the A5 Linux host")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(TOOLKIT_HOME "/usr/local/Ascend/cann-9.2.0" CACHE PATH "Actual CANN root")
set(BM25_VENDOR_DIR "" CACHE PATH "Four aliases created by prepare_sharded_runtime.py")
find_path(ACL_INCLUDE acl/acl.h PATHS "${TOOLKIT_HOME}/include" NO_DEFAULT_PATH)
find_path(META_INCLUDE aclnn/acl_meta.h PATHS "${TOOLKIT_HOME}/include" "${TOOLKIT_HOME}/include/aclnnop" NO_DEFAULT_PATH)
find_library(ACL_LIBRARY ascendcl PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
find_library(NNOP_LIBRARY nnopbase PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
if(NOT ACL_INCLUDE OR NOT META_INCLUDE OR NOT ACL_LIBRARY OR NOT NNOP_LIBRARY OR NOT BM25_VENDOR_DIR)
    message(FATAL_ERROR "Missing actual CANN runtime/headers or BM25_VENDOR_DIR")
endif()
set(API_INCLUDES "")
foreach(vendor bm25 bm25_accumulate bm25_topk bm25_topk_merge)
    if(NOT EXISTS "${BM25_VENDOR_DIR}/${vendor}/op_api/lib/libcust_opapi.so")
        message(FATAL_ERROR "Actual installed API library missing: ${vendor}")
    endif()
    list(APPEND API_INCLUDES "${BM25_VENDOR_DIR}/${vendor}/op_api/include")
endforeach()
foreach(header bm25/aclnn_bm25_contribution.h bm25_accumulate/aclnn_bm25_accumulate.h bm25_topk/aclnn_bm25_top_k.h bm25_topk_merge/aclnn_bm25_top_k_merge.h)
    string(REPLACE "/" "/op_api/include/" actual_header "${header}")
    if(NOT EXISTS "${BM25_VENDOR_DIR}/${actual_header}")
        message(FATAL_ERROR "Actual generated header missing: ${actual_header}")
    endif()
endforeach()
add_executable(bm25_sharded_benchmark sharded_benchmark.cpp)
target_include_directories(bm25_sharded_benchmark PRIVATE "${ACL_INCLUDE}" "${META_INCLUDE}" ${API_INCLUDES})
target_link_libraries(bm25_sharded_benchmark PRIVATE "${NNOP_LIBRARY}" "${ACL_LIBRARY}" ${CMAKE_DL_LIBS})
target_compile_options(bm25_sharded_benchmark PRIVATE -Wall -Wextra -O2)
set_target_properties(bm25_sharded_benchmark PROPERTIES BUILD_RPATH "${TOOLKIT_HOME}/lib64")
message(STATUS "Four-stage generated headers: ${API_INCLUDES}")
message(STATUS "Four absolute API paths and all eight symbol owners are checked at runtime")
