cmake_minimum_required(VERSION 3.17)
project(bm25_search_benchmark LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Build on A5 Linux using GCC/Clang and actual CANN headers")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenMP REQUIRED COMPONENTS CXX)
set(TOOLKIT_HOME "/usr/local/Ascend/cann-9.2.0" CACHE PATH "Actual CANN toolkit root")
set(BM25_VENDOR_DIR "" CACHE PATH "Actual installed vendors parent")
find_path(ACL_INCLUDE acl/acl.h PATHS "${TOOLKIT_HOME}/include" NO_DEFAULT_PATH)
find_path(META_INCLUDE aclnn/acl_meta.h PATHS "${TOOLKIT_HOME}/include" "${TOOLKIT_HOME}/include/aclnnop" NO_DEFAULT_PATH)
find_library(ACL_LIBRARY ascendcl PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
find_library(NNOP_LIBRARY nnopbase PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
if(NOT ACL_INCLUDE OR NOT META_INCLUDE OR NOT ACL_LIBRARY OR NOT NNOP_LIBRARY OR NOT BM25_VENDOR_DIR)
    message(FATAL_ERROR "Actual CANN headers/libraries or BM25_VENDOR_DIR missing")
endif()
set(API_INCLUDES "")
foreach(pair bm25_query_score/aclnn_bm25_query_score.h bm25_topk_radix_collect/aclnn_bm25_top_k.h bm25_topk_merge/aclnn_bm25_top_k_merge.h)
    string(REPLACE "/" "/op_api/include/" header "${pair}")
    if(NOT EXISTS "${BM25_VENDOR_DIR}/${header}")
        message(FATAL_ERROR "Actual generated header missing: ${header}")
    endif()
    get_filename_component(parent "${BM25_VENDOR_DIR}/${header}" DIRECTORY)
    list(APPEND API_INCLUDES "${parent}")
endforeach()
set(COMMON_SOURCES npu_prototype/search_benchmark_host/CMakeLists.txt npu_prototype/search_benchmark_host/benchmark_common.h
    npu_prototype/search_host/search_inputs.h)
set(CPU_SOURCES ${COMMON_SOURCES} npu_prototype/search_benchmark_host/cpu_benchmark.cpp
    cpu_reference/index_search/cpu_index_search.h cpu_reference/topk_document_shards.h)
set(NPU_SOURCES ${COMMON_SOURCES} npu_prototype/search_benchmark_host/npu_benchmark.cpp
    npu_prototype/search_host/npu_search.h npu_prototype/resident_chain_host/resident_chain_common.h
    npu_prototype/query_score_host/query_score_runtime.h)
foreach(backend cpu npu)
    string(TOUPPER "${backend}" upper)
    add_executable(bm25_${backend}_search_benchmark ${backend}_benchmark.cpp)
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${backend}")
    set(record "#pragma once\n#include <cstddef>\nstatic constexpr const char *BM25_BENCH_SOURCES[][2] = {\n")
    foreach(source IN LISTS ${upper}_SOURCES)
        set(path "${CMAKE_CURRENT_SOURCE_DIR}/../../${source}")
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
        file(SHA256 "${path}" sha)
        string(APPEND record "{\"${source}\",\"${sha}\"},\n")
    endforeach()
    string(APPEND record "};\nstatic constexpr std::size_t BM25_BENCH_SOURCE_COUNT=sizeof(BM25_BENCH_SOURCES)/sizeof(BM25_BENCH_SOURCES[0]);\n")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${backend}/compiled_benchmark_sources.h" "${record}")
    target_include_directories(bm25_${backend}_search_benchmark PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${backend}")
endforeach()
target_link_libraries(bm25_cpu_search_benchmark PRIVATE OpenMP::OpenMP_CXX)
target_compile_options(bm25_cpu_search_benchmark PRIVATE -O3 -march=native -ffp-contract=off -fno-fast-math -Wall -Wextra)
target_include_directories(bm25_npu_search_benchmark PRIVATE "${ACL_INCLUDE}" "${META_INCLUDE}" ${API_INCLUDES})
target_link_libraries(bm25_npu_search_benchmark PRIVATE "${NNOP_LIBRARY}" "${ACL_LIBRARY}" ${CMAKE_DL_LIBS})
target_compile_options(bm25_npu_search_benchmark PRIVATE -O2 -ffp-contract=off -fno-fast-math -Wall -Wextra)
set_target_properties(bm25_npu_search_benchmark PROPERTIES BUILD_RPATH "${TOOLKIT_HOME}/lib64")
message(STATUS "Reuse accepted Search implementations unchanged; only add alternating-query host-wall-clock measurement")
message(STATUS "Index/capacity cold once; query uploads and Top-K downloads remain inside every NPU measured Search")
