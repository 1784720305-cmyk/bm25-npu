cmake_minimum_required(VERSION 3.17)
project(bm25_query_id_search LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Build on the A5 Linux host against actual installed generated headers")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(TOOLKIT_HOME "/usr/local/Ascend/cann-9.2.0" CACHE PATH "Actual CANN toolkit root")
set(BM25_VENDOR_DIR "" CACHE PATH "Actual installed vendors parent")
find_path(ACL_INCLUDE acl/acl.h PATHS "${TOOLKIT_HOME}/include" NO_DEFAULT_PATH)
find_path(META_INCLUDE aclnn/acl_meta.h PATHS "${TOOLKIT_HOME}/include" "${TOOLKIT_HOME}/include/aclnnop" NO_DEFAULT_PATH)
find_library(ACL_LIBRARY ascendcl PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
find_library(NNOP_LIBRARY nnopbase PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
if(NOT ACL_INCLUDE OR NOT META_INCLUDE OR NOT ACL_LIBRARY OR NOT NNOP_LIBRARY OR NOT BM25_VENDOR_DIR)
    message(FATAL_ERROR "Actual CANN headers/libraries or BM25_VENDOR_DIR missing")
endif()
set(API_INCLUDES "")
foreach(pair bm25_query_score/aclnn_bm25_query_score.h bm25_topk_radix_collect/aclnn_bm25_top_k.h bm25_topk_merge/aclnn_bm25_top_k_merge.h)
    string(REPLACE "/" "/op_api/include/" header "${pair}")
    if(NOT EXISTS "${BM25_VENDOR_DIR}/${header}")
        message(FATAL_ERROR "Actual generated header missing: ${header}")
    endif()
    get_filename_component(parent "${BM25_VENDOR_DIR}/${header}" DIRECTORY)
    list(APPEND API_INCLUDES "${parent}")
    message(STATUS "Actual generated header: ${BM25_VENDOR_DIR}/${header}")
endforeach()
foreach(vendor bm25_query_score bm25_topk_radix_collect bm25_topk_merge)
    if(NOT EXISTS "${BM25_VENDOR_DIR}/${vendor}/op_api/lib/libcust_opapi.so")
        message(FATAL_ERROR "Actual installed API library missing: ${vendor}")
    endif()
endforeach()
add_executable(bm25_search_runner search_runner.cpp)
target_include_directories(bm25_search_runner PRIVATE "${ACL_INCLUDE}" "${META_INCLUDE}" ${API_INCLUDES})
target_link_libraries(bm25_search_runner PRIVATE "${NNOP_LIBRARY}" "${ACL_LIBRARY}" ${CMAKE_DL_LIBS})
target_compile_options(bm25_search_runner PRIVATE -Wall -Wextra -O2 -fno-fast-math -ffp-contract=off)
set(SEARCH_SOURCES
    npu_prototype/search_host/CMakeLists.txt
    npu_prototype/search_host/search_inputs.h
    npu_prototype/search_host/npu_search.h
    npu_prototype/search_host/search_runner.cpp
    npu_prototype/resident_chain_host/resident_chain_common.h
    npu_prototype/query_score_host/query_score_runtime.h)
set(RECORD "#pragma once\n#include <cstddef>\nstatic constexpr const char *BM25_SEARCH_SOURCES[][2] = {\n")
foreach(source IN LISTS SEARCH_SOURCES)
    set(path "${CMAKE_CURRENT_SOURCE_DIR}/../../${source}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
    file(SHA256 "${path}" source_sha)
    string(APPEND RECORD "{\"${source}\",\"${source_sha}\"},\n")
endforeach()
string(APPEND RECORD "};\nstatic constexpr std::size_t BM25_SEARCH_SOURCE_COUNT = sizeof(BM25_SEARCH_SOURCES)/sizeof(BM25_SEARCH_SOURCES[0]);\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/compiled_search_sources.h" "${RECORD}")
target_include_directories(bm25_search_runner PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
set_target_properties(bm25_search_runner PROPERTIES BUILD_RPATH "${TOOLKIT_HOME}/lib64")
include(CTest)
if(BUILD_TESTING)
    add_executable(bm25_search_input_tests search_input_tests.cpp)
    target_compile_options(bm25_search_input_tests PRIVATE -Wall -Wextra -O2)
    add_test(NAME search_public_inputs COMMAND bm25_search_input_tests)
endif()
message(STATUS "Public Search: raw grouped CPU IDs -> packing/upload/score/Top-K/merge/sync/download -> CPU result")
message(STATUS "New query-ID Search boundary diagnostic; requires actual device correctness; no CPU fallback or speedup claim")
