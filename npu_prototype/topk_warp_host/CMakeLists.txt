cmake_minimum_required(VERSION 3.17)
project(bm25_topk_warp_diagnostic LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Build on the A5 Linux host using the installed generated header")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(TOOLKIT_HOME "/usr/local/Ascend/cann-9.2.0" CACHE PATH "Actual toolkit root")
set(BM25_VENDOR_DIR "" CACHE PATH "Existing installed vendors; also supplies shared host helper headers")
find_path(ACL_INCLUDE acl/acl.h PATHS "${TOOLKIT_HOME}/include" NO_DEFAULT_PATH)
find_path(META_INCLUDE aclnn/acl_meta.h PATHS "${TOOLKIT_HOME}/include" "${TOOLKIT_HOME}/include/aclnnop" NO_DEFAULT_PATH)
find_library(ACL_LIBRARY ascendcl PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
find_library(NNOP_LIBRARY nnopbase PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
if(NOT ACL_INCLUDE OR NOT META_INCLUDE OR NOT ACL_LIBRARY OR NOT NNOP_LIBRARY)
    message(FATAL_ERROR "Actual CANN headers/runtime libraries are missing")
endif()
set(API_INCLUDES "")
foreach(pair bm25/aclnn_bm25_contribution.h bm25_accumulate/aclnn_bm25_accumulate.h bm25_topk_warp/aclnn_bm25_top_k.h)
    string(REPLACE "/" "/op_api/include/" actual_header "${pair}")
    if(NOT EXISTS "${BM25_VENDOR_DIR}/${actual_header}")
        message(FATAL_ERROR "Actual generated header missing: ${actual_header}")
    endif()
    get_filename_component(header_dir "${BM25_VENDOR_DIR}/${actual_header}" DIRECTORY)
    list(APPEND API_INCLUDES "${header_dir}")
endforeach()
if(NOT EXISTS "${BM25_VENDOR_DIR}/bm25_topk_warp/op_api/lib/libcust_opapi.so")
    message(FATAL_ERROR "Installed warp API library is missing")
endif()
add_executable(bm25_topk_warp_runner warp_runner.cpp)
target_include_directories(bm25_topk_warp_runner PRIVATE "${ACL_INCLUDE}" "${META_INCLUDE}" ${API_INCLUDES})
target_link_libraries(bm25_topk_warp_runner PRIVATE "${NNOP_LIBRARY}" "${ACL_LIBRARY}" ${CMAKE_DL_LIBS})
target_compile_options(bm25_topk_warp_runner PRIVATE -Wall -Wextra -O2)
set_target_properties(bm25_topk_warp_runner PROPERTIES BUILD_RPATH "${TOOLKIT_HOME}/lib64")
message(STATUS "Actual warp header: ${BM25_VENDOR_DIR}/bm25_topk_warp/op_api/include/aclnn_bm25_top_k.h")
message(STATUS "Shared runtime/tensor helpers only; loads and executes ONLY the explicit warp API library")
