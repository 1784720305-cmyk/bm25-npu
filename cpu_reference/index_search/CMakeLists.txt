cmake_minimum_required(VERSION 3.16)
project(bm25_cpu_indexed_search LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenMP REQUIRED COMPONENTS CXX)
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "CPU Search needs GCC/Clang with the documented FP32 compiler flags")
endif()
add_executable(bm25_cpu_index_search_runner cpu_search_runner.cpp)
target_link_libraries(bm25_cpu_index_search_runner PRIVATE OpenMP::OpenMP_CXX)
target_compile_options(bm25_cpu_index_search_runner PRIVATE -O3 -march=native -ffp-contract=off -fno-fast-math -Wall -Wextra)
set(SEARCH_SOURCES
    cpu_reference/index_search/CMakeLists.txt
    cpu_reference/index_search/cpu_index_search.h
    cpu_reference/index_search/cpu_search_runner.cpp
    cpu_reference/topk_document_shards.h
    npu_prototype/search_host/search_inputs.h)
set(RECORD "#pragma once\n#include <cstddef>\nstatic constexpr const char *BM25_CPU_SEARCH_SOURCES[][2] = {\n")
foreach(source IN LISTS SEARCH_SOURCES)
    set(path "${CMAKE_CURRENT_SOURCE_DIR}/../../${source}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
    file(SHA256 "${path}" source_sha)
    string(APPEND RECORD "{\"${source}\",\"${source_sha}\"},\n")
endforeach()
string(APPEND RECORD "};\nstatic constexpr std::size_t BM25_CPU_SEARCH_SOURCE_COUNT = sizeof(BM25_CPU_SEARCH_SOURCES)/sizeof(BM25_CPU_SEARCH_SOURCES[0]);\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/compiled_cpu_search_sources.h" "${RECORD}")
target_include_directories(bm25_cpu_index_search_runner PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
include(CTest)
if(BUILD_TESTING)
    add_executable(bm25_cpu_index_search_tests cpu_index_search_tests.cpp)
    target_link_libraries(bm25_cpu_index_search_tests PRIVATE OpenMP::OpenMP_CXX)
    target_compile_options(bm25_cpu_index_search_tests PRIVATE -O3 -march=native -ffp-contract=off -fno-fast-math -Wall -Wextra)
    add_test(NAME cpu_index_search_correctness COMMAND bm25_cpu_index_search_tests)
endif()
message(STATUS "CPU Search: same raw grouped word IDs, resident raw-TF index, ordered FP32 scoring, own document-sharded Top-K")
message(STATUS "Standard C++/OpenMP only; no CANN, BM25S, Faiss or IndexSDK runtime dependency")
