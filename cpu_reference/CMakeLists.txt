cmake_minimum_required(VERSION 3.16)
project(bm25_cpu_reference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenMP REQUIRED COMPONENTS CXX)
add_executable(bm25_cpu_runner cpu_runner.cpp)
target_link_libraries(bm25_cpu_runner PRIVATE OpenMP::OpenMP_CXX)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(bm25_cpu_runner PRIVATE -O3 -march=native -ffp-contract=off -fno-fast-math -Wall -Wextra)
else()
    message(FATAL_ERROR "This initial CPU baseline requires GCC/Clang with the documented FP32 flags")
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(bm25_cpu_topk_tests topk_document_shards_test.cpp)
    target_link_libraries(bm25_cpu_topk_tests PRIVATE OpenMP::OpenMP_CXX)
    target_compile_options(bm25_cpu_topk_tests PRIVATE -O3 -march=native -ffp-contract=off -fno-fast-math -Wall -Wextra)
    add_test(NAME cpu_document_sharded_topk COMMAND bm25_cpu_topk_tests)
endif()
