cmake_minimum_required(VERSION 3.17)
project(bm25_final LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BM25_BUILD_NPU "Build the final CANN host and device-check entry" ON)
set(TOOLKIT_HOME "/usr/local/Ascend/cann-9.2.0" CACHE PATH "CANN toolkit")
set(BM25_VENDOR_DIR "" CACHE PATH "Aliases to final fused and tree Merge vendors")
find_package(OpenMP REQUIRED COMPONENTS CXX)
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 message(FATAL_ERROR "Use GCC/Clang with the declared FP32 options")
endif()
# Keep computational source paths/bytes identical to accepted binaries.
set(CORE_SOURCES
 common/large_search_plan.h common/fixed_query_packet.h common/shard_directory.h
 cpu_reference/topk_document_shards.h cpu_reference/index_search/cpu_index_search.h
 cpu_reference/directory_search/cpu_directory_search.h
 npu_prototype/search_host/search_inputs.h
 npu_prototype/search_benchmark_host/benchmark_common.h
 npu_prototype/search_benchmark_host/cpu_benchmark.cpp
 npu_prototype/graph_search_host/cpu_directory_benchmark.cpp
 npu_prototype/fused_chain_host/fused_runtime.h npu_prototype/fused_search_host/fused_api.h
 npu_prototype/graph_search_host/planned_search.h
 npu_prototype/large_search_host/benchmark.cpp npu_prototype/large_search_host/large_search.h
 npu_prototype/large_search_host/merge_check.h CMakeLists.txt)
set(RECORD "#pragma once\n#include <cstddef>\nstatic constexpr const char *BM25_BENCH_SOURCES[][2] = {\n")
foreach(source IN LISTS CORE_SOURCES)
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
 file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${source}" sha)
 string(APPEND RECORD "{\"${source}\",\"${sha}\"},\n")
endforeach()
string(APPEND RECORD "};\nstatic constexpr std::size_t BM25_BENCH_SOURCE_COUNT=sizeof(BM25_BENCH_SOURCES)/sizeof(BM25_BENCH_SOURCES[0]);\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/compiled_benchmark_sources.h" "${RECORD}")
add_executable(bm25_cpu_search_benchmark npu_prototype/search_benchmark_host/cpu_benchmark.cpp)
add_executable(bm25_directory_cpu_benchmark npu_prototype/graph_search_host/cpu_directory_benchmark.cpp)
foreach(target bm25_cpu_search_benchmark bm25_directory_cpu_benchmark)
 target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
 target_link_libraries(${target} PRIVATE OpenMP::OpenMP_CXX)
 target_compile_options(${target} PRIVATE -O3 -march=native -ffp-contract=off -fno-fast-math -Wall -Wextra)
endforeach()
include(CTest)
if(BUILD_TESTING)
 add_executable(bm25_cpu_contract_check cpu_reference/index_search/cpu_index_search_tests.cpp)
 target_link_libraries(bm25_cpu_contract_check PRIVATE OpenMP::OpenMP_CXX)
 target_compile_options(bm25_cpu_contract_check PRIVATE -O3 -ffp-contract=off -fno-fast-math)
 add_test(NAME cpu_contract COMMAND bm25_cpu_contract_check)
endif()
if(BM25_BUILD_NPU)
 if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "NPU targets require A5 Linux; CPU-only checks use -DBM25_BUILD_NPU=OFF")
 endif()
 find_path(ACL_INCLUDE acl/acl.h PATHS "${TOOLKIT_HOME}/include" NO_DEFAULT_PATH)
 find_path(META_INCLUDE aclnn/acl_meta.h PATHS "${TOOLKIT_HOME}/include" "${TOOLKIT_HOME}/include/aclnnop" NO_DEFAULT_PATH)
 find_library(ACL_LIBRARY ascendcl PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
 find_library(NNOP_LIBRARY nnopbase PATHS "${TOOLKIT_HOME}/lib64" NO_DEFAULT_PATH)
 if(NOT ACL_INCLUDE OR NOT META_INCLUDE OR NOT ACL_LIBRARY OR NOT NNOP_LIBRARY OR NOT BM25_VENDOR_DIR)
  message(FATAL_ERROR "Actual CANN and final runtime aliases required")
 endif()
 foreach(pair bm25_fused_score_topk/aclnn_bm25_fused_score_top_k.h bm25_topk_merge/aclnn_bm25_top_k_merge.h)
  string(REPLACE "/" "/op_api/include/" header "${pair}")
  if(NOT EXISTS "${BM25_VENDOR_DIR}/${header}")
   message(FATAL_ERROR "Actual generated API missing: ${header}")
  endif()
  get_filename_component(parent "${BM25_VENDOR_DIR}/${header}" DIRECTORY)
  list(APPEND API_INCLUDES "${parent}")
 endforeach()
 add_executable(bm25_large_search_benchmark npu_prototype/large_search_host/benchmark.cpp)
 add_executable(bm25_fused_simd_selector_check npu_prototype/fused_simd_check_host/selector_check.cpp)
 file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/npu_prototype/fused_simd_check_host/selector_check.cpp" CHECK_SHA)
 target_compile_definitions(bm25_fused_simd_selector_check PRIVATE BM25_SIMD_CHECK_SHA="${CHECK_SHA}")
 foreach(target bm25_large_search_benchmark bm25_fused_simd_selector_check)
  target_include_directories(${target} PRIVATE "${ACL_INCLUDE}" "${META_INCLUDE}" ${API_INCLUDES}
   "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/npu_prototype/fused_chain_host")
  target_link_libraries(${target} PRIVATE "${NNOP_LIBRARY}" "${ACL_LIBRARY}" ${CMAKE_DL_LIBS})
  target_compile_options(${target} PRIVATE -O3 -ffp-contract=off -fno-fast-math -Wall -Wextra)
  set_target_properties(${target} PROPERTIES BUILD_RPATH "${TOOLKIT_HOME}/lib64")
 endforeach()
endif()
