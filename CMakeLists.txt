cmake_minimum_required(VERSION 3.17)
project(bm25_indexsdk LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(BM25_BUILD_NPU "Build CANN host benchmarks and device checks" ON)
option(BM25_BUILD_BENCHMARKS "Build standalone CPU/NPU benchmark programs" ON)
set(TOOLKIT_HOME "/usr/local/Ascend/cann-9.2.0" CACHE PATH "CANN toolkit")
set(BM25_VENDOR_DIR "" CACHE PATH "Fused and Merge operator vendors")
set(BM25_INDEXSDK_SOURCE "" CACHE PATH "IndexSDK source root, enables the SDK facade and adapter tests")
set(BM25_INDEXSDK_LIBRARY "" CACHE FILEPATH "Actual IndexSDK host library, enables shared-resource device acceptance")
set(BM25_SECUREC_INCLUDE_DIR "" CACHE PATH "Directory containing securec.h for the SDK resource headers")
set(BM25_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(BM25_INSTALL_LIBDIR "host/lib" CACHE STRING "IndexSDK-compatible library install directory")
set(BM25_INSTALL_INCLUDEDIR "${CMAKE_INSTALL_INCLUDEDIR}/ascendfaiss" CACHE STRING "BM25 header root")
add_subdirectory(feature_retrieval/src/ascendfaiss)
include(cmake/Install.cmake)
